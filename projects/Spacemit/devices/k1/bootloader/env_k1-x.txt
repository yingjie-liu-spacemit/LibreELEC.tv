// K1 LibreELEC u-boot env overlay - imported from bootfs env_k1-x.txt
// Boot chain: SPL -> OpenSBI -> U-Boot (already on disk), this env controls
// the kernel boot: Image + dtb from the "bootfs" partition, built-in initramfs.

// Console
console=ttyS0,115200
bootdelay=0

// Boot kernel via booti (raw Image, no external initrd - initramfs is built in)
// Addresses match the vendor U-Boot layout (CONFIG_FASTBOOT_BUF_ADDR / DTB_LOAD_ADDR)
knl_name=Image
kernel_addr_r=0x8000000
dtb_addr=0x31000000

// Common boot args (no splash/plymouth, LibreELEC brings its own)
// systemd.debug_shell=1: root shell on DEBUG_TTY (/dev/ttyS0) for bringup
commonargs=setenv bootargs earlyprintk clk_ignore_unused swiotlb=65536 quiet workqueue.default_affinity_scope=system systemd.debug_shell=1

set_console=if test "${console}" != "";then setenv bootargs "${bootargs}" console=${console};fi
add_bootarg=run set_console

// mmc (SD/eMMC) boot args: add boot= so the LibreELEC initramfs can find SYSTEM
set_mmc_args=setenv bootargs "${bootargs}" rootwait rootfstype=ext4 boot=/dev/mmcblk${boot_devnum}p${bootfs_part}

// rootfs partition (name "rootfs") - used as LibreELEC /storage
set_root_arg=run get_rootfs_uuid; setenv bootargs "${bootargs}" root=UUID=${rootfs_uuid} disk=UUID=${rootfs_uuid};

// DTB selection based on product_name from board EEPROM (vendor logic)
dtb_env=if test -n "${product_name}"; then echo "match dtb by product_name: ${product_name}.dtb"; setenv dtb_name ${product_name}.dtb; fi;
detect_dtb=echo "product_name: ${product_name}"; run dtb_env; echo "select ${dtb_name} to load";

loadknl=echo "Loading kernel..."; load ${bootfs_devname} ${boot_devnum}:${bootfs_part} ${kernel_addr_r} ${knl_name};
loaddtb=echo "Loading dtb..."; if load ${bootfs_devname} ${boot_devnum}:${bootfs_part} ${dtb_addr} ${dtb_name}; then else setenv dtb_addr ""; fi;

// no external ramdisk: ramdisk_combo "-" -> kernel built-in initramfs
ramdisk_combo=-

start_kernel=booti ${kernel_addr_r} ${ramdisk_combo} ${dtb_addr};

boot_kernel=run set_root_arg; run detect_dtb; run loadknl; run loaddtb; run start_kernel;

mmc_boot=echo "Try to boot from ${bootfs_devname}${boot_devnum} ..."; run commonargs; run add_bootarg; run set_mmc_args; run boot_kernel; echo "########### boot failed by default config, check your boot config #############"

autoboot=if test ${boot_device} = mmc; then run mmc_boot; else echo "boot_device ${boot_device} not supported, falling back to mmc"; run mmc_boot; fi;

bootcmd=run autoboot; echo "run autoboot"
